{{- /* [params.theme], normalised — the palette name and the token overrides. Returns dict "palette" "css" . The palette is the whole point of the guard: baseof.html stamps it into `` and head.html builds a favicon filename out of it, so an unchecked value is an attribute and a path built from config. Both come back from here already reduced to one of five literals. Cache it per language at the call sites — head.html and baseof.html both want it on every page, the answer cannot change within a language, and [languages.pt.params.theme] is a different answer from its en counterpart behind the same path. Uncached, the warnings below would be recomputed once per page for a result that is identical every time. Nothing here may print: this returns a value, so every action is whitespace-trimmed. An untrimmed line breaks `return`. */ -}} {{- $known := slice "lime" "amber" "cyberpunk" "ice" "mono" -}} {{- /* The keys sites write, mapped to the custom properties terminal.css reads. They are spelled lowercase here because that is how they arrive: Hugo lowercases every config key recursively, so `accentDim` in the TOML is `accentdim` by the time it reaches site.Params. Sites should still write camelCase — it is what the rest of the theme's params use, and TOML bare keys cannot carry the `-` the CSS side needs. Both spellings work; only this side has to know that. */ -}} {{- $tokens := dict "bg" "--bg" "surface" "--surface" "surface2" "--surface-2" "surface3" "--surface-3" "border" "--border" "bordersoft" "--border-soft" "borderfaint" "--border-faint" "text" "--text" "muted" "--muted" "muted2" "--muted-2" "dim" "--dim" "dim2" "--dim-2" "soft" "--soft" "prose" "--prose" "accent" "--accent" "accentdim" "--accent-dim" "danger" "--danger" "codekey" "--code-key" "codetype" "--code-type" "codestr" "--code-str" "codenum" "--code-num" -}} {{- /* The same names as a list, for the warning to print. Derived rather than written out a second time, so it cannot drift from the map above — and `range` over a map walks it in sorted key order, which is what keeps the message stable between builds. */ -}} {{- $tokenNames := slice -}} {{- range $name, $_ := $tokens -}}{{- $tokenNames = $tokenNames | append $name -}}{{- end -}} {{- $theme := partial "params-table.html" (dict "value" site.Params.theme "path" "[params.theme]") -}} {{- /* ---- palette ---- */ -}} {{- $palette := "lime" -}} {{- $raw := partial "params-scalar.html" (dict "value" $theme.palette "path" "[params.theme] palette") -}} {{- with $raw -}} {{- $name := lower (printf "%v" .) -}} {{- if in $known $name -}} {{- $palette = $name -}} {{- else -}} {{- warnf "terminal-mono: [params.theme] palette = %q is not a palette this theme ships — using %q. Available: %s" . "lime" (delimit $known ", ") -}} {{- end -}} {{- end -}} {{- /* ---- per-token overrides ---- */ -}} {{- $colors := partial "params-table.html" (dict "value" $theme.colors "path" "[params.theme.colors]") -}} {{- $decls := slice -}} {{- $unknown := slice -}} {{- $rejected := slice -}} {{- range $key, $value := $colors -}} {{- $prop := index $tokens $key -}} {{- if not $prop -}} {{- $unknown = $unknown | append $key -}} {{- else -}} {{- $v := printf "%v" $value -}} {{- /* A value goes into a stylesheet as written, so it is checked before it gets there rather than trusted. Without `;` and `}` a value cannot end the declaration it sits in, and without `:` it cannot start another one — which is the whole of the injection this could otherwise be. What survives is still wide enough for anything a color is: #rgb, #rrggbbaa, rgb(0 0 0 / 40%), oklch(70% .2 150), a bare keyword. `url(` is excluded because a token that fetches something is not a color, whatever it is spelled like. */ -}} {{- if and (findRE `^[#A-Za-z0-9(),./% -]+$` $v) (not (in (lower $v) "url(")) (le (len $v) 64) -}} {{- $decls = $decls | append (printf "%s:%s" $prop $v) -}} {{- else -}} {{- $rejected = $rejected | append $key -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Both of these are typos that would otherwise do nothing at all, quietly — the failure mode params-list.html already counts its dropped entries for. A site that sets `acccent` and sees no warning concludes the feature is broken; one that sees this fixes a letter. */ -}} {{- with $unknown -}} {{- warnf "terminal-mono: [params.theme.colors] ignored %d unknown key(s): %s — Hugo lowercases config keys, so compare these against: %s" (len .) (delimit . ", ") (delimit $tokenNames ", ") -}} {{- end -}} {{- with $rejected -}} {{- warnf "terminal-mono: [params.theme.colors] ignored %d value(s) that are not colors: %s — write a hex code like \"#b6ff3c\" or a CSS color function" (len .) (delimit . ", ") -}} {{- end -}} {{- return dict "palette" $palette "css" (delimit $decls ";") -}}
