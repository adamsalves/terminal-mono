{{- /* Say out loud what t.html quietly papers over. The fallback keeps an untranslated site readable; it does not make it translated, and nothing else on the page reports that. So this counts the keys the reader's language cannot answer and warns once per language — the rule sections.html set and the params-* guards enforce: a build never fails on configuration, and configuration never fails in silence. The count comes from the translation *files*, not from calling i18n on every key, because the resolved value cannot tell a translation from a fallback: Hugo substitutes the site's defaultContentLanguage for a missing key, so `i18n "about"` in an untranslated `es` returns the English string, not "". Asking the value therefore saw nothing wrong with any language but the default one — a site with `[languages.es]` and no i18n/es.toml, which is the setup the README walks you through, warned about nothing at all. Both halves of Hugo's merge are covered. The theme's own tables come from data/ (see i18n-en.html and the theme's hugo.toml); the site's come from the union filesystem, which returns the site's file when there is one. A site that translates ten of the theme's 49 Portuguese strings is reported as missing none of them, because the theme's own pt table is read here too. Called from head.html through partialCached keyed on the language, so it is one pass per language per build, not per page. Prints nothing. */ -}} {{- $lang := site.Language.Lang -}} {{- $en := partialCached "i18n-en.html" "en" -}} {{- $have := dict -}} {{- with index hugo.Data "terminal_mono_i18n" $lang -}} {{- if reflect.IsMap . -}}{{- $have = merge $have . -}}{{- end -}} {{- end -}} {{- /* Hugo accepts any of these for a translation table, so look for all of them. try() because the file is the site's to write and a malformed one must warn, not abort the build. */ -}} {{- range $ext := slice "toml" "yaml" "yml" "json" -}} {{- $path := printf "i18n/%s.%s" $lang $ext -}} {{- if os.FileExists $path -}} {{- with try (transform.Unmarshal (os.ReadFile $path)) -}} {{- if .Err -}} {{- warnf "terminal-mono: could not parse %s (%s), so this build cannot say which UI strings it translates. Anything missing still falls back to English." $path .Err -}} {{- else if reflect.IsMap .Value -}} {{- $have = merge $have .Value -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $missing := slice -}} {{- range $key, $_ := $en -}} {{- $answered := false -}} {{- with index $have $key -}} {{- if reflect.IsMap . -}} {{- /* A key whose every plural form is "" is a blank label on the page, which is the state this whole path exists to make visible. */ -}} {{- range $text := . -}}{{- if $text -}}{{- $answered = true -}}{{- end -}}{{- end -}} {{- else -}} {{- $answered = true -}} {{- end -}} {{- end -}} {{- if not $answered -}}{{- $missing = $missing | append $key -}}{{- end -}} {{- end -}} {{- with $missing -}} {{- if eq (len .) (len $en) -}} {{- warnf "terminal-mono: no UI strings for language %q — the interface is falling back to English. Copy the theme's i18n/en.toml to i18n/%s.toml in your site and translate it." $lang $lang -}} {{- else -}} {{- /* Named, because a list of keys is what you act on and a count is not. Capped, because a file with three strings in it would otherwise print the other 46 into the build log every run. */ -}} {{- $sorted := sort . -}} {{- $shown := $sorted -}} {{- $tail := "" -}} {{- if gt (len $sorted) 12 -}} {{- $shown = first 12 $sorted -}} {{- $tail = printf " and %d more" (sub (len $sorted) 12) -}} {{- end -}} {{- warnf "terminal-mono: %d of %d UI strings are missing for language %q and fall back to English — add them to i18n/%s.toml: %s%s" (len .) (len $en) $lang $lang (delimit $shown ", ") $tail -}} {{- end -}} {{- end -}}
