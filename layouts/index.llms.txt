{{- /* llms.txt — https://llmstxt.org

       An index an answer engine can read in one request instead of crawling:
       what the site is, and every page on it as a linked list with a line of
       context each. The format is markdown on purpose — it is the one structure
       every model already parses — and it is deliberately small. llms-full.txt
       is where the content goes; this file is the map to it.

       The `## Optional` heading is not a placeholder. The spec gives that one
       section a defined meaning: a reader working with a short context window
       may skip it. Section index pages are exactly that — they carry no prose
       of their own, only links to the posts already listed above.

       One per language. Hugo renders a home page's output formats per language,
       so a bilingual site gets /llms.txt and /pt/llms.txt with each one listing
       its own posts, and the header states which language it is.

       The file is built as a list of lines and joined at the end rather than
       written as literal text with trim markers. Plain text has no forgiving
       renderer behind it: in HTML a stray newline is whitespace nobody sees,
       here it is a blank line in the middle of a list, and every `{{- ` in a
       template like this is one more place to put it. Appending "" is a blank
       line stated on purpose.

       Everything below reads the same guarded params head.html does, and for
       the same reason: a scalar where a table belongs would abort the build,
       and this renders on the home page of every language. */ -}}
{{- $siteName := or site.Params.title site.Title -}}
{{- $hero := partialCached "params-table.html" (dict "value" site.Params.hero "path" "[params.hero]") "hero" site.Language.Lang -}}
{{- $siteDesc := partialCached "params-scalar.html" (dict "value" site.Params.description "path" "[params.description]") "description" site.Language.Lang -}}
{{- $posts := where site.RegularPages "Type" "blogs" -}}
{{- if not $posts }}{{ $posts = site.RegularPages }}{{ end -}}

{{- $out := slice (printf "# %s" $siteName) "" -}}
{{- with $siteDesc }}{{ $out = $out | append (printf "> %s" (. | plainify | strings.TrimSpace)) "" }}{{ end -}}
{{- with $hero.subtitle }}{{ $out = $out | append (. | plainify | strings.TrimSpace) "" }}{{ end -}}

{{- $out = $out | append (printf "- Language: %s" (or site.Language.Locale site.Language.Lang)) -}}
{{- $out = $out | append (printf "- Home: %s" .Permalink) -}}
{{- with .OutputFormats.Get "llmsfull" -}}
  {{- $out = $out | append (printf "- Full text of every post, in one file: %s" .Permalink) -}}
{{- end -}}
{{- range .Translations -}}
  {{- $out = $out | append (printf "- Also published in %s: %s" (or .Language.Locale .Language.Lang) .Permalink) -}}
{{- end -}}

{{- with $posts -}}
  {{- $out = $out | append "" (printf "## %s" (partial "t.html" "blog" | title)) "" -}}
  {{- range . -}}
    {{- /* The link goes to the markdown twin when there is one. That is what the
           spec asks for and what a model following it actually wants — the
           source rather than a page it has to strip the chrome off — and it
           costs the citation nothing, because the first metadata line inside
           that file is the canonical URL. A site that publishes no MARKDOWN
           output gets the HTML permalink here instead. */ -}}
    {{- $href := .Permalink -}}
    {{- with .OutputFormats.Get "markdown" }}{{ $href = .Permalink }}{{ end -}}
    {{- $line := printf "- [%s](%s)" .Title $href -}}
    {{- /* .Summary comes back with the trailing newline its <p> had, and
           plainify keeps it. One stray newline in a line-joined file is a blank
           line in the middle of the list, so every value that reaches a line
           here is trimmed rather than trusted. */ -}}
    {{- with or .Description .Summary -}}
      {{- $line = printf "%s: %s" $line (. | plainify | strings.TrimSpace | truncate 200) -}}
    {{- end -}}
    {{- $out = $out | append $line -}}
  {{- end -}}
{{- end -}}

{{- with site.Taxonomies.tags -}}
  {{- $out = $out | append "" (printf "## %s" (partial "t.html" "tags" | title)) "" -}}
  {{- range $name, $taxonomy := . -}}
    {{- with $taxonomy.Page -}}
      {{- $n := len $taxonomy.Pages -}}
      {{- $out = $out | append (printf "- [%s](%s): %d post%s" $name .Permalink $n (cond (eq $n 1) "" "s")) -}}
    {{- end -}}
  {{- end -}}
{{- end -}}

{{- with where site.Pages "Kind" "section" -}}
  {{- $out = $out | append "" "## Optional" "" -}}
  {{- range . -}}
    {{- $out = $out | append (printf "- [%s](%s): index of the posts listed above" .Title .Permalink) -}}
  {{- end -}}
{{- end -}}
{{ delimit $out "\n" }}
